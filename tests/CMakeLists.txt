cmake_minimum_required(VERSION 3.16)
project(SkellTuneDSPTests LANGUAGES CXX)
enable_testing()
file(GLOB sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/RealtimeAutotune_*.cpp")
foreach(source IN LISTS sources)
  get_filename_component(name "${source}" NAME_WE)
  add_executable("${name}" "${source}")
  target_compile_features("${name}" PRIVATE cxx_std_17)
  target_include_directories("${name}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../Source")
  if(name STREQUAL "RealtimeAutotune_real_vocal_octave")
    add_test(NAME "${name}" COMMAND "${name}" "${CMAKE_CURRENT_SOURCE_DIR}/assets/real_voice_octave_200hz.wav")
  else()
    add_test(NAME "${name}" COMMAND "${name}")
  endif()
endforeach()
